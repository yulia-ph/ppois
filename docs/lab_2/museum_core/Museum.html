324    def count_visitors_today(self, access_records) -> int:
325        """Count today's public-zone visits from access records.
326
327        Only records that were granted, occurred today, and target the
328        public zone are counted.
329
330        Args:
331            access_records: An iterable of AccessControl-like objects.
332                Each must expose `.is_granted`, `.zone`, and `.time`.
333
334        Returns:
335            The number of visitors admitted to the public zone today.
336        """
337        today = date.today()
338        count = 0
339        for record in access_records:
340            if not record.is_granted:
341                continue
342            if record.zone != "public":
343                continue
344            if record.time.date() != today:
345                continue
346            count += 1
347        return count
