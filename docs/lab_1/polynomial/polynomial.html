 46    def __str__(self):
 47        """@public Return a human-readable string representation of the polynomial."""
 48        if not self._coefficients:
 49            return "0"
 50
 51        res = []
 52        for i, c in enumerate(self._coefficients):
 53            if c == 0:
 54                continue
 55            elif i == 0:
 56                res.append(f"{c}")
 57            elif i == 1:
 58                if abs(c) == 1:
 59                    res.append("-x" if c < 0 else "x")

167    def __divmod__(self, other):
168        """@public Return (quotient, remainder) of polynomial division.
169
170            Returns:
171                A tuple (Q, R) such that self == other * Q + R
172                and degree(R) < degree(other).
173            Raises:
174                TypeError: if other is not a Polynomial.
175                ZeroDivisionError: if other is the zero polynomial.
176            """
177        if not isinstance(other, Polynomial):
178            raise TypeError(f"Cannot divide Polynomial by {type(other)}")
179        if other._coefficients == [0]:
180            raise ZeroDivisionError("Division by the zero polynomial")
181
182        dividend = list(self._coefficients)
183        divisor = other._coefficients
184
185        if len(dividend) < len(divisor):
186            return self._from_internal([0]), self._from_internal(dividend)
187
188        quotient = [0] * (len(dividend) - len(divisor) + 1)
189
190        while len(dividend) >= len(divisor) and not (len(dividend) == 1 and dividend[0] == 0):
191
192            coef = dividend[-1] / divisor[-1]
193            degree_diff = len(dividend) - len(divisor)
194            quotient[degree_diff] = coef
195
196            for i in range(len(divisor)):
197                dividend[degree_diff + i] -= coef * divisor[i]
198
199            while len(dividend) > 1 and dividend[-1] == 0:
200                dividend.pop()
201
202        return self._from_internal(quotient), self._from_internal(dividend)
203
